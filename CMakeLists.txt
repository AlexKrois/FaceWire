cmake_minimum_required(VERSION 3.24)
project(camoutlines VERSION 0.2.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype currently targets Windows x64 with MSVC.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DEPS "${CMAKE_SOURCE_DIR}/.deps")
set(OBS_INCLUDE_DIR "${DEPS}/obs-studio-30.2.3/libobs" CACHE PATH "OBS headers")
set(OBS_LIBRARY "${DEPS}/obs.lib" CACHE FILEPATH "OBS import library")
set(MEDIAPIPE_INCLUDE_DIR "${DEPS}/mediapipe-0.10.32" CACHE PATH "MediaPipe source root")
add_library(cam-detector STATIC src/detector.cpp)
target_include_directories(cam-detector PUBLIC src PRIVATE "${MEDIAPIPE_INCLUDE_DIR}")
target_compile_options(cam-detector PRIVATE /W4 /utf-8 /EHsc)
add_library(camoutlines MODULE src/plugin.cpp)
target_include_directories(camoutlines PRIVATE "${OBS_INCLUDE_DIR}")
target_link_libraries(camoutlines PRIVATE cam-detector "${OBS_LIBRARY}")
target_compile_options(camoutlines PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(camoutlines PROPERTIES PREFIX "")
install(TARGETS camoutlines RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
install(DIRECTORY data/ DESTINATION data/obs-plugins/camoutlines)
install(FILES "${DEPS}/mediapipe/mediapipe/tasks/c/libmediapipe.dll" "${DEPS}/face_landmarker.task" DESTINATION data/obs-plugins/camoutlines)
install(FILES THIRD_PARTY.md LICENSE DESTINATION data/obs-plugins/camoutlines)
include(CTest)
if(BUILD_TESTING)
  add_executable(geometry-test tests/geometry.cpp)
  target_include_directories(geometry-test PRIVATE src)
  add_test(NAME geometry COMMAND geometry-test)
  add_executable(detector-test tests/detector.cpp)
  target_link_libraries(detector-test PRIVATE cam-detector)
  add_test(NAME detector COMMAND detector-test "${DEPS}/mediapipe/mediapipe/tasks/c/libmediapipe.dll" "${DEPS}/face_landmarker.task")
  add_executable(obs-render-test tests/obs-render.cpp)
  target_include_directories(obs-render-test PRIVATE "${OBS_INCLUDE_DIR}")
  target_link_libraries(obs-render-test PRIVATE "${OBS_LIBRARY}")
endif()
